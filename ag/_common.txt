70| def learner_section(previous, heading, template):
71|     """Return the learner-owned tail of a generated file so a rerun never drops prose.
72| 
73|     The tail starts at `heading`; if the learner removed that heading, the whole old text is kept
74|     below a fresh template instead of being overwritten.
75|     """
76|     if heading in previous:
77|         return previous[previous.index(heading):]
78|     if not previous.strip():
79|         return template
80|     return template + ("\n## Bản cũ\n\nLệnh không thấy tiêu đề `%s` nên giữ nguyên bản cũ dưới đây. "
81|                        "Chuyển phần bạn viết lên mục trên rồi xoá mục này.\n\n%s" % (heading.lstrip("# "), previous))

88| def ensure_late_templates(base):
89|     """Copy P4–P6 templates into submission/ once r1_craft reference is open; never overwrite."""
90|     sub = base / "submission"
91|     if not (sub / "r1_craft" / "reference.txt").is_file():
92|         return []
93|     created = []
94|     for name in LATE_TEMPLATES:
95|         source, target = base / "assets" / "templates" / name, sub / name
96|         if source.is_file() and not target.exists():
97|             target.write_bytes(source.read_bytes())
98|             created.append(name)
99|     return created
