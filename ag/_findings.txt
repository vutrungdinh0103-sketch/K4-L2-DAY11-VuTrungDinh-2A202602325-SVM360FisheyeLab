16| def read_rows(path):
17|     if not path.is_file():
18|         return []
19|     with path.open(encoding="utf-8-sig", newline="") as stream:
20|         return [row for row in csv.DictReader(stream) if any((v or "").strip() for v in row.values())]

23| def append_rows(path, rows):
24|     path.parent.mkdir(parents=True, exist_ok=True)
25|     existing = read_rows(path)
26|     keys = {(r.get("round"), r.get("frame"), r.get("object_ref"), r.get("what")) for r in existing}
27|     if not path.is_file():
28|         with path.open("w", encoding="utf-8", newline="") as stream:
29|             csv.writer(stream, lineterminator="\r\n").writerow(HEADER)
30|     with path.open("a", encoding="utf-8", newline="") as stream:
31|         writer = csv.DictWriter(stream, HEADER, extrasaction="ignore", lineterminator="\r\n")
32|         for row in rows:
33|             key = tuple(row.get(k) for k in ("round", "frame", "object_ref", "what"))
34|             if key not in keys:
35|                 writer.writerow({name: row.get(name, "") for name in HEADER})
36|                 keys.add(key)
